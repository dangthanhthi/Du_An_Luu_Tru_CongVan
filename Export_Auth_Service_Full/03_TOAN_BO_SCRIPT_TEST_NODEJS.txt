async function httpGet(url, token = null, customHeaders = {}) {
    const headers = { ...customHeaders };
    if (token && !headers['Authorization']) headers['Authorization'] = `Bearer ${token}`;
    const start = Date.now();
    const res = await fetch(url, { headers });
    const duration = Date.now() - start;
    const ct = res.headers.get('content-type') || '';
    let resBody;
    const rawText = await res.text();
    try { resBody = JSON.parse(rawText); } catch { resBody = rawText; }
    return { status: res.status, body: resBody, duration, headers: res.headers };
}

async function httpPost(url, jsonBody, token = null, customHeaders = {}) {
    const headers = { 'Content-Type': 'application/json', ...customHeaders };
    if (token && !headers['Authorization']) headers['Authorization'] = `Bearer ${token}`;
    const start = Date.now();
    const res = await fetch(url, { method: 'POST', headers, body: typeof jsonBody === 'string' ? jsonBody : JSON.stringify(jsonBody) });
    const duration = Date.now() - start;
    let resBody;
    const rawText = await res.text();
    try { resBody = JSON.parse(rawText); } catch { resBody = rawText; }
    return { status: res.status, body: resBody, duration, headers: res.headers };
}

function assert(name, actual, expected) {
    const passed = String(actual) === String(expected);
    return { name, passed, actual: String(actual), expected: String(expected) };
}

function assertTruthy(name, value) {
    return { name, passed: !!value, actual: String(!!value), expected: 'true' };
}

function recordTest(id, category, name, desc, method, url, expectedStatus, actualStatus, passed, responseBody, durationMs, assertions, error = null) {
    const result = { id, category, name, description: desc, method, url, expectedStatus, actualStatus, passed, responseBody, durationMs, assertions, error };
    results.push(result);
    const icon = passed ? '✅' : '❌';
    console.log(`${icon} [${passed ? 'PASS' : 'FAIL'}] ${id}: ${name} (${durationMs}ms)`);
    if (!passed && error) console.log(`   └── ERROR: ${error}`);
    return result;
}
